# 1. Install dependencies
cd client && npm install        # adds swiper
cd ../server && npm install     # swaps better-sqlite3 → pg

# 2. Create your PostgreSQL database (local)
psql -U postgres -c "CREATE DATABASE vega;"

# 3. Create server/.env from the example
cp server/.env.example server/.env
# Then edit DATABASE_URL:
DATABASE_URL=postgresql://postgres:YOUR_PASSWORD@localhost:5432/vega

# 4. (Cloud hosts like Render/Railway/Supabase)
# Set DATABASE_URL in their dashboard env vars panel.
# Uncomment the ssl line in server/src/db.js if required:
#   ssl: { rejectUnauthorized: false }

# 5. Start
npm run dev   # from repo root (if concurrently is set up), or:
cd server && npm run dev
cd client && npm run dev
